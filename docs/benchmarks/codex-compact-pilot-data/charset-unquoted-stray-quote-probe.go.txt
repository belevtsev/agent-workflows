package main
import("encoding/json";"net/http";"net/http/httptest";"os";"runtime";"strings";"github.com/go-chi/chi/v5/middleware")
func main(){
type result struct{Case string `json:"case"`;Header string `json:"header"`;Allowed []string `json:"allowed"`;BodyBytes int64 `json:"body_bytes"`;Status int `json:"status"`}
rows:=[]result{}
for _,test:=range []struct{name,header string;allowed []string}{
{"legacy_unquoted_stray_quote_utf8","text/plain; note=foo\"bar; charset=utf-8",[]string{"UTF-8"}},
{"legacy_unquoted_stray_quote_disallowed_latin1","text/plain; note=foo\"bar; charset=latin-1",[]string{"UTF-8",""}},
}{
r:=httptest.NewRequest(http.MethodPost,"http://example.invalid/",strings.NewReader("body"));r.Header.Set("Content-Type",test.header)
w:=httptest.NewRecorder();handler:=middleware.ContentCharset(append([]string(nil),test.allowed...)...)(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){w.WriteHeader(http.StatusNoContent)}));handler.ServeHTTP(w,r)
rows=append(rows,result{test.name,test.header,test.allowed,r.ContentLength,w.Code})
}
if err:=json.NewEncoder(os.Stdout).Encode(map[string]any{"go_version":runtime.Version(),"results":rows});err!=nil{panic(err)}
}
